function loadContent(){
  const [m,maps,n,ps]=await Promise.all([
    sb.from('fztv_site_matches').select('*').order('starts_at',{ascending:false}),
    sb.from('fztv_match_maps').select('*').order('map_order',{ascending:true}),
    sb.from('fztv_news').select('*').order('created_at',{ascending:false}),
    sb.from('fztv_player_match_stats').select('*')
  ]);
  if(m.error||maps.error||n.error)console.warn('FZTV content:',m.error||maps.error||n.error);
  matchesCache=(m.data||[]).map(match=>{
    const matchMaps=(maps.data||[]).filter(x=>x.match_id===match.id)
      .map(x=>({name:x.map_name,a:x.score_a,b:x.score_b}));
    const mapWinsA=matchMaps.filter(x=>x.a!=null&&x.b!=null&&x.a>x.b).length;
    const mapWinsB=matchMaps.filter(x=>x.a!=null&&x.b!=null&&x.b>x.a).length;
    const score=String(match.result_overall||'').match(/(\d+)\s*[:–-]\s*(\d+)/);
    return {...match,event_name:match.title||'FragZoneTV',maps:matchMaps,
      score_a:score?Number(score[1]):mapWinsA,score_b:score?Number(score[2]):mapWinsB};
  });
  newsCache=n.data||[];
  playerMatchStats=ps.data||[];
  if(ps.error)console.warn('Player stats:',ps.error);
  contentLoaded=true;render();
}

async function adminWrite(table,payload,id){
  if(table==='fztv_site_matches'){
    const {maps=[],score_a=0,score_b=0,event_name='',...fields}=payload;
    const record={...fields,title:event_name,result_overall:String(score_a)+':'+String(score_b)};
    const q=id?sb.from(table).update(record).eq('id',id):sb.from(table).insert(record);
    const {data,error}=await q.select('id').single();
    if(error)throw error;
    const matchId=data.id;
    if(id){
      const del=await sb.from('fztv_match_maps').delete().eq('match_id',matchId);
      if(del.error)throw del.error;
    }
    if(maps.length){
      const insert=await sb.from('fztv_match_maps').insert(maps.map((x,i)=>({
        match_id:matchId,map_name:x.name,map_order:i+1,score_a:x.a,score_b:x.b
      })));
      if(insert.error)throw insert.error;
    }
  }else{
    if(table==='fztv_news'){
      payload={...payload,content:payload.body,published:true};
      delete payload.body;
    }
    const q=id?sb.from(table).update(payload).eq('id',id):sb.from(table).insert(payload);
    const {error}=await q;
    if(error)throw error;
  }
  await loadContent();
}
