}

async function loadContent(){

    sb.from('fztv_site_matches').select('*').order('starts_at',{ascending:false}),
    sb.from('fztv_match_maps').select('*').order('map_order',{ascending:true}),
    sb.from('fztv_news').select('*').order('created_at',{ascending:false}),
    sb.from('fztv_player_match_stats').select('*')
  ]);
  if(m.error||maps.error||n.error)console.warn('FZTV content:',m.error||maps.error||n.error);
  matchesCache=(m.data||[]).map(match=>{
    const matchMaps=(maps.data||[]).filter(x=>x.match_id===match.id)
      .map(x=>({name:x.map_name,a:x.score_a,b:x.score_b}));
    const mapWinsA=matchMaps.filter(x=>x.a!=null&&x.b!=null&&x.a>x.b).length;
    const mapWinsB=matchMaps.filter(x=>x.a!=null&&x.b!=null&&x.b>x.a).length;
    const score=String(match.result_overall||'').match(/(\d+)\s*[:–-]\s*(\d+)/);
    return {...match,event_name:match.title||'FragZoneTV',maps:matchMaps,
      score_a:score?Number(score[1]):mapWinsA,score_b:score?Number(score[2]):mapWinsB};
  });
  newsCache=n.data||[];
  playerMatchStats=ps.data||[];
  if(ps.error)console.warn('Player stats:',ps.error);
  contentLoaded=true;render();
}

      .eq('id',sessionUser.id).maybeSingle();

    const ban=await sb.rpc('fztv_my_ban');
    if(ban.error)console.warn('Cannot check account ban:',ban.error);
    if(ban.data){
      await sb.auth.signOut();sessionUser=null;profile=null;
      alert('Аккаунт заблокирован. Причина: '+(ban.data.reason||'не указана'));
    }else{
      const adm=await sb.rpc('fztv_is_admin');
      if(adm.error)console.warn('Admin role check:',adm.error);

        const own=await sb.rpc('fztv_is_owner');

  }
