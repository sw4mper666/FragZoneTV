function refreshAuth(){
  const {data:{user},error}=await sb.auth.getUser();
  sessionUser=error?null:user;profile=null;adminVerified=false;ownerVerified=false;
  if(sessionUser){
    const p=await sb.from('profiles').select('id,login,email,role,points,created_at,avatar_url,bio,banner_color')
      .eq('id',sessionUser.id).maybeSingle();
    profile=p.data||{login:sessionUser.user_metadata?.login||sessionUser.email?.split('@')[0]};
    const ban=await sb.rpc('fztv_my_ban');
    if(ban.error)console.warn('Cannot check account ban:',ban.error);
    if(ban.data){
      await sb.auth.signOut();sessionUser=null;profile=null;
      alert('Аккаунт заблокирован. Причина: '+(ban.data.reason||'не указана'));
    }else{
      const adm=await sb.rpc('fztv_is_admin');
      if(adm.error)console.warn('Admin role check:',adm.error);
      adminVerified=adm.data===true;
      if(adminVerified){
        const own=await sb.rpc('fztv_is_owner');
        ownerVerified=own.data===true;
      }
    }
  }
  render();
  if(adminVerified && location.hash.startsWith('#/admin')){
    loadUsers().catch(e=>console.error('Пользователи:',e));
  }
}
